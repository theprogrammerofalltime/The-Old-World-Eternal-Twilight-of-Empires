country_event = {
	id = towe_balkan.1
	title = towe_balkan.1.t
	desc = towe_balkan.1.d
	picture = GFX_report_event_generic_riot

	fire_only_once = yes
	major = yes

	trigger = {
		tag = JAS
		date > 1906.5.1
		date < 1906.6.15
		NOT = { has_country_flag = towe_balkan_revolt_started }
	}

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		set_country_flag = towe_balkan_revolt_started
	}

	option = {
		name = towe_balkan.1.a
		custom_effect_tooltip = towe_balkan_revolt_tt

		load_focus_tree = JAS_balkan_crisis_focus


		every_other_country = {
			limit = {
				OR = {
					tag = BRC
					tag = GAU
					tag = HAS
					tag = PRG
					tag = PLC
					tag = NUO
					tag = IBE
					tag = KSE
					tag = IME
				}
				is_subject = no
			}
			country_event = { id = towe_balkan.2 days = 3 }
		}

		# The Balkan Crisis is a conventional interstate war, not a civil war.
		# Give the Buffer States the Bulgarian territories before hostilities.
		JSE = { transfer_state = 48 }
		JSE = { transfer_state = 801 }
		JSE = { transfer_state = 211 }
		JSE = { transfer_state = 212 }
		JSE = {
			set_country_flag = towe_balkan_buffer_states_united
			country_event = { id = towe_balkan.3 days = 1 }
		}

		add_stability = -0.12
		add_war_support = 0.08
		add_political_power = -75
		set_global_flag = towe_balkan_crisis_active
	}
}
