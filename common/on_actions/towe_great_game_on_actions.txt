# The two-bloc struggle is initialized once and updated in JAS country scope.
on_actions = {
    on_startup = {
        effect = {
            JAS = {
                set_variable = { var = towe_great_game_tension value = 20 }
                set_variable = { var = towe_great_game_influence value = 0 }
            }
            PRG = {
                set_variable = { var = towe_great_game_influence value = 0 }
            }
        }
    }
    on_monthly_JAS = {
        effect = {
            if = {
                limit = {
                    check_variable = { towe_great_game_tension > 0 }
                    has_war = no
                    PRG = { has_war = no }
                }
                add_to_variable = { var = towe_great_game_tension value = -1 }
                clamp_variable = { var = towe_great_game_tension min = 0 max = 100 }
            }
            if = {
                limit = {
                    check_variable = { towe_great_game_tension > 74 }
                    NOT = { has_country_flag = towe_great_game_crisis_notified }
                }
                set_country_flag = towe_great_game_crisis_notified
                country_event = { id = towe_great_game.1 days = 1 }
            }
            if = {
                limit = { check_variable = { towe_great_game_tension < 40 } }
                clr_country_flag = towe_great_game_crisis_notified
            }
        }
    }
}
