	clear_array = towe_policy_effectiveness_effect_array
	multiply_temp_variable = { towe_policy_effectiveness_temp_math = 0.01 } #you need to be in percents pls

	for_each_loop = { #there is no size() function in pdx code, so i'm doing this like that
		array = towe_policy_effectiveness_max_array
		value = max_value #while i'm here, might as well save some time
		index = i

		set_temp_variable = { effect = max_value }
		subtract_from_temp_variable = { effect = towe_policy_effectiveness_min_array^i }
		multiply_temp_variable = { effect = towe_policy_effectiveness_temp_math }		#defined in prev scope
		add_to_temp_variable = { effect = towe_policy_effectiveness_min_array^i }

		add_to_array = { towe_policy_effectiveness_effect_array = effect }
	}

	clear_array = towe_policy_effectiveness_max_array
	clear_array = towe_policy_effectiveness_min_array
}
